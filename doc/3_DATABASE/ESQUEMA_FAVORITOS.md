# Sistema de Favoritos - Paseo del Comercio

## 📋 Resumen Ejecutivo

Este documento describe la implementación del sistema de favoritos para tiendas y productos en la aplicación móvil de Paseo del Comercio. El sistema permite a los usuarios guardar tiendas y productos como favoritos para acceder rápidamente a ellos.

---

## 🗄️ Base de Datos

### Nuevas Tablas

#### 1. `tienda_favorito`

Almacena las tiendas que un usuario ha marcado como favoritas.

| Columna | Tipo | Nullable | Default | Descripción |
|---------|------|----------|---------|-------------|
| id | bigint | NO | generated by default as identity | ID único del registro |
| usuario_id | bigint | NO | - | FK al usuario que marcó favorito |
| tienda_id | bigint | NO | - | FK a la tienda marcada como favorita |
| fecha_creacion | timestamp with time zone | NO | now() | Fecha cuando se creó el favorito |

**Constraints:**
- `tienda_favorito_pkey` PRIMARY KEY (id)
- `tienda_favorito_usuario_tienda_unique` UNIQUE (usuario_id, tienda_id)
- `tienda_favorito_usuario_id_fkey` FOREIGN KEY (usuario_id) REFERENCES usuario(id)
- `tienda_favorito_tienda_id_fkey` FOREIGN KEY (tienda_id) REFERENCES tienda(id)

**Índices:**
- `idx_tienda_favorito_usuario` (usuario_id)
- `idx_tienda_favorito_tienda` (tienda_id)

#### 2. `producto_favorito`

Almacena los productos que un usuario ha marcado como favoritos.

| Columna | Tipo | Nullable | Default | Descripción |
|---------|------|----------|---------|-------------|
| id | bigint | NO | generated by default as identity | ID único del registro |
| usuario_id | bigint | NO | - | FK al usuario que marcó favorito |
| producto_id | bigint | NO | - | FK al producto marcado como favorito |
| fecha_creacion | timestamp with time zone | NO | now() | Fecha cuando se creó el favorito |

**Constraints:**
- `producto_favorito_pkey` PRIMARY KEY (id)
- `producto_favorito_usuario_producto_unique` UNIQUE (usuario_id, producto_id)
- `producto_favorito_usuario_id_fkey` FOREIGN KEY (usuario_id) REFERENCES usuario(id)
- `producto_favorito_producto_id_fkey` FOREIGN KEY (producto_id) REFERENCES producto(id)

**Índices:**
- `idx_producto_favorito_usuario` (usuario_id)
- `idx_producto_favorito_producto` (producto_id)

### SQL de Creación

```sql
-- Tabla de favoritos de tiendas
CREATE TABLE tienda_favorito (
  id BIGSERIAL PRIMARY KEY,
  usuario_id BIGINT NOT NULL REFERENCES usuario(id),
  tienda_id BIGINT NOT NULL REFERENCES tienda(id),
  fecha_creacion TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  CONSTRAINT tienda_favorito_usuario_tienda_unique UNIQUE (usuario_id, tienda_id)
);

CREATE INDEX idx_tienda_favorito_usuario ON tienda_favorito(usuario_id);
CREATE INDEX idx_tienda_favorito_tienda ON tienda_favorito(tienda_id);

-- Tabla de favoritos de productos
CREATE TABLE producto_favorito (
  id BIGSERIAL PRIMARY KEY,
  usuario_id BIGINT NOT NULL REFERENCES usuario(id),
  producto_id BIGINT NOT NULL REFERENCES producto(id),
  fecha_creacion TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  CONSTRAINT producto_favorito_usuario_producto_unique UNIQUE (usuario_id, producto_id)
);

CREATE INDEX idx_producto_favorito_usuario ON producto_favorito(usuario_id);
CREATE INDEX idx_producto_favorito_producto ON producto_favorito(producto_id);
```

### Funciones RPC (Opcional)

```sql
-- Obtener tiendas favoritas del usuario
CREATE OR REPLACE FUNCTION get_tiendas_favoritas(p_usuario_id BIGINT)
RETURNS TABLE (
  id BIGINT,
  tienda_id BIGINT,
  nombre_tienda VARCHAR,
  descripcion TEXT,
  logo_url TEXT,
  fecha_creacion TIMESTAMPTZ,
  fecha_favorito TIMESTAMPTZ
) AS $$
BEGIN
  RETURN QUERY
  SELECT 
    tf.id,
    t.id AS tienda_id,
    t.nombre_tienda,
    t.descripcion,
    t.logo_url,
    t.fecha_creacion,
    tf.fecha_creacion AS fecha_favorito
  FROM tienda_favorito tf
  JOIN tienda t ON tf.tienda_id = t.id
  WHERE tf.usuario_id = p_usuario_id
  ORDER BY tf.fecha_creacion DESC;
END;
$$ LANGUAGE plpgsql;

-- Obtener productos favoritos del usuario
CREATE OR REPLACE FUNCTION get_productos_favoritos(p_usuario_id BIGINT)
RETURNS TABLE (
  id BIGINT,
  producto_id BIGINT,
  nombre VARCHAR,
  descripcion TEXT,
  precio_base NUMERIC,
  imagen_url TEXT,
  nombre_tienda VARCHAR,
  fecha_creacion TIMESTAMPTZ,
  fecha_favorito TIMESTAMPTZ
) AS $$
BEGIN
  RETURN QUERY
  SELECT 
    pf.id,
    p.id AS producto_id,
    p.nombre,
    p.descripcion,
    p.precio_base,
    ip.url_imagen AS imagen_url,
    t.nombre_tienda,
    p.fecha_creacion,
    pf.fecha_creacion AS fecha_favorito
  FROM producto_favorito pf
  JOIN producto p ON pf.producto_id = p.id
  JOIN tienda t ON p.tienda_id = t.id
  LEFT JOIN LATERAL (
    SELECT url_imagen 
    FROM imagen_productos 
    WHERE id_producto = p.id AND tipo_imagen = 'principal' AND activa = true 
    LIMIT 1
  ) ip ON true
  WHERE pf.usuario_id = p_usuario_id
  ORDER BY pf.fecha_creacion DESC;
END;
$$ LANGUAGE plpgsql;

-- Toggle favorito de tienda
CREATE OR REPLACE FUNCTION toggle_tienda_favorito(
  p_usuario_id BIGINT,
  p_tienda_id BIGINT
)
RETURNS BOOLEAN AS $$
DECLARE
  existe_favorito BOOLEAN;
BEGIN
  -- Verificar si ya es favorito
  EXISTS(
    SELECT 1 
    FROM tienda_favorito 
    WHERE usuario_id = p_usuario_id AND tienda_id = p_tienda_id
  );
  
  IF EXISTS THEN
    -- Eliminar favorito
    DELETE FROM tienda_favorito 
    WHERE usuario_id = p_usuario_id AND tienda_id = p_tienda_id;
    RETURN FALSE;
  ELSE
    -- Crear favorito
    INSERT INTO tienda_favorito (usuario_id, tienda_id) 
    VALUES (p_usuario_id, p_tienda_id);
    RETURN TRUE;
  END IF;
END;
$$ LANGUAGE plpgsql;

-- Toggle favorito de producto
CREATE OR REPLACE FUNCTION toggle_producto_favorito(
  p_usuario_id BIGINT,
  p_producto_id BIGINT
)
RETURNS BOOLEAN AS $$
BEGIN
  IF EXISTS (
    SELECT 1 
    FROM producto_favorito 
    WHERE usuario_id = p_usuario_id AND producto_id = p_producto_id
  ) THEN
    DELETE FROM producto_favorito 
    WHERE usuario_id = p_usuario_id AND producto_id = p_producto_id;
    RETURN FALSE;
  ELSE
    INSERT INTO producto_favorito (usuario_id, producto_id) 
    VALUES (p_usuario_id, p_producto_id);
    RETURN TRUE;
  END IF;
END;
$$ LANGUAGE plpgsql;
```

### Políticas para tienda_favorito

```sql
-- Usuario solo puede ver sus propios favoritos
CREATE POLICY "Usuarios ven sus propios favoritos de tiendas"
ON tienda_favorito FOR SELECT
USING (usuario_id = auth.uid());

-- Usuario solo puede insertar sus propios favoritos
CREATE POLICY "Usuarios crean favoritos de tiendas"
ON tienda_favorito FOR INSERT
WITH CHECK (usuario_id = auth.uid());

-- Usuario solo puede eliminar sus propios favoritos
CREATE POLICY "Usuarios eliminan favoritos de tiendas"
ON tienda_favorito FOR DELETE
USING (usuario_id = auth.uid());

-- No permitir actualización
CREATE POLICY "No permite actualizar favoritos de tiendas"
ON tienda_favorito FOR UPDATE
USING (false);
```

### Función helper para verificar favorito

```sql
-- Función para verificar si es favorito
CREATE OR REPLACE FUNCTION is_favorite(
  p_usuario_id BIGINT,
  p_tienda_id BIGINT DEFAULT NULL,
  p_producto_id BIGINT DEFAULT NULL
)
RETURNS BOOLEAN AS $$
BEGIN
  IF p_tienda_id IS NOT NULL THEN
    RETURN EXISTS (
      SELECT 1 FROM tienda_favorito
      WHERE usuario_id = p_usuario_id AND tienda_id = p_tienda_id
    );
  ELSIF p_producto_id IS NOT NULL THEN
    RETURN EXISTS (
      SELECT 1 FROM producto_favorito
      WHERE usuario_id = p_usuario_id AND producto_id = p_producto_id
    );
  END IF;
  RETURN FALSE;
END;
$$ LANGUAGE plpgsql;
```
